	LOAD IT

	QUICK START
		local Window = Nauru:CreateWindow({ Title = "NAURU", Subtitle = "HUB", Key = Enum.KeyCode.RightShift, Theme = "ember" })

local DERIVE = {
	secTitle = function(t) return t.text:Lerp(t.sub, 0.3) end,
	track = function(t) return t.line:Lerp(t.sub, 0.06) end,
	divider = function(t) return t.line:Lerp(t.sub, 0.35) end,
	hoverLine = function(t) return t.line:Lerp(t.accent, 0.4) end,
	hoverLine2 = function(t) return t.line:Lerp(t.accent, 0.25) end,
	hoverField = function(t) return t.field:Lerp(t.text, 0.06) end,
	accentHover = function(t) return t.accent:Lerp(Color3.new(1, 1, 1), 0.15) end,

local function finishTheme(t)
	for k, f in pairs(DERIVE) do t[k] = f(t) end
	t.white, t.black = Color3.new(1, 1, 1), Color3.new(0, 0, 0)
	return t
end

local C = table.clone(THEMES.ember) -- the live palette
local bound, hooks = {}, {}          -- themed instances / callbacks run after a theme change

local function bind(inst, map)
	bound[#bound + 1] = { inst, map }
	for prop, key in pairs(map) do inst[prop] = C[key] end
	return inst
end

local function rgbStr(c) return string.format("rgb(%d,%d,%d)", math.floor(c.R * 255 + 0.5), math.floor(c.G * 255 + 0.5), math.floor(c.B * 255 + 0.5)) end

local function create(class, props, parent)
	local o = Instance.new(class)
	for k, v in pairs(props) do o[k] = v end
	o.Parent = parent
	return o
end

local function tween(o, t, props, style, dir)
	local tw = TweenService:Create(o, TweenInfo.new(t, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out), props)
	tw:Play()
	return tw
end

local function frame(parent, props)
	if props.BackgroundColor3 == nil and props.BackgroundTransparency == nil then props.BackgroundTransparency = 1 end
	props.BorderSizePixel = 0
	return create("Frame", props, parent)
end

local function corner(o, r) return create("UICorner", { CornerRadius = UDim.new(0, r) }, o) end

local function stroke(o, key, th, tr)
	local s = create("UIStroke", { Thickness = th or 1, Transparency = tr or 0, ApplyStrokeMode = Enum.ApplyStrokeMode.Border }, o)
	bind(s, { Color = key })
	return s
end

local function label(parent, text, size, key, bold, props)
	local p = {
		BackgroundTransparency = 1, BorderSizePixel = 0, Text = text, TextSize = size, Font = bold and FONT_BOLD or FONT,
		TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd,
	}
	for k, v in pairs(props or {}) do p[k] = v end
	local l = create("TextLabel", p, parent)
	bind(l, { TextColor3 = key })
	return l
end

local function button(parent, props)
	local p = { AutoButtonColor = false, Text = "", BorderSizePixel = 0, BackgroundTransparency = 1, Font = FONT, TextSize = 12 }
	for k, v in pairs(props or {}) do p[k] = v end
	return create("TextButton", p, parent)
end

local function isPress(i) return i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch end
local function isMove(i) return i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch end

local function bindDrag(hit, onDrag)
	local dragging = false
	local function update(input)
		local a, s = hit.AbsolutePosition, hit.AbsoluteSize
		onDrag(math.clamp((input.Position.X - a.X) / math.max(s.X, 1), 0, 1), math.clamp((input.Position.Y - a.Y) / math.max(s.Y, 1), 0, 1))
	end
	hit.InputBegan:Connect(function(input) if isPress(input) then dragging = true update(input) end end)
	UserInputService.InputChanged:Connect(function(input) if dragging and isMove(input) then update(input) end end)
	UserInputService.InputEnded:Connect(function(input) if isPress(input) then dragging = false end end)
end

local function isInside(gui, pos)
	local a, s = gui.AbsolutePosition, gui.AbsoluteSize
	return pos.X >= a.X and pos.X <= a.X + s.X and pos.Y >= a.Y and pos.Y <= a.Y + s.Y
end

local gui, toastHolder

local function ensureGui()
	if gui and gui.Parent then return gui end
	local targets = {}
	pcall(function() if gethui then table.insert(targets, gethui()) end end)
	pcall(function() table.insert(targets, game:GetService("CoreGui")) end)
	table.insert(targets, Players.LocalPlayer:WaitForChild("PlayerGui"))

	for _, t in ipairs(targets) do
		pcall(function()
			local old = t:FindFirstChild("NauruUI")
			if old then old:Destroy() end
		end)
	end
	gui = Instance.new("ScreenGui")
	gui.Name = "NauruUI"
	gui.ResetOnSpawn = false
	gui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
	gui.IgnoreGuiInset = true
	gui.DisplayOrder = 999
	pcall(function() if syn and syn.protect_gui then syn.protect_gui(gui) end end)
	local parented = false
	for _, t in ipairs(targets) do
		if pcall(function() gui.Parent = t end) and gui.Parent == t then parented = true break end
	end
	if not parented then gui.Parent = Players.LocalPlayer:WaitForChild("PlayerGui") end

	toastHolder = frame(gui, { Name = "Toasts", AnchorPoint = Vector2.new(1, 1), Position = UDim2.new(1, -16, 1, -16), Size = UDim2.fromOffset(250, 400) })
	create("UIListLayout", { Padding = UDim.new(0, 6), VerticalAlignment = Enum.VerticalAlignment.Bottom, HorizontalAlignment = Enum.HorizontalAlignment.Right, SortOrder = Enum.SortOrder.LayoutOrder }, toastHolder)
	return gui
end

function Nauru:Notify(cfg)
	ensureGui()
	cfg = type(cfg) == "table" and cfg or { Text = tostring(cfg) }
	local wrap = frame(toastHolder, { Size = UDim2.fromOffset(240, 0), AutomaticSize = Enum.AutomaticSize.Y })
	local card = frame(wrap, { Position = UDim2.fromOffset(270, 0), Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, BackgroundTransparency = 0 })
	bind(card, { BackgroundColor3 = "panel" })
	corner(card, 8)
	stroke(card, "accent", 1, 0.7)
	create("UIPadding", { PaddingTop = UDim.new(0, 9), PaddingBottom = UDim.new(0, 10), PaddingLeft = UDim.new(0, 12), PaddingRight = UDim.new(0, 12) }, card)
	create("UIListLayout", { Padding = UDim.new(0, 3), SortOrder = Enum.SortOrder.LayoutOrder }, card)
	local tl = label(card, cfg.Title or "Notification", 12, "accent", true, { Size = UDim2.new(1, 0, 0, 16), LayoutOrder = 1 })
	if cfg.Text and cfg.Text ~= "" then
		label(card, cfg.Text, 11, "sub", false, { Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, TextWrapped = true, TextTruncate = Enum.TextTruncate.None, LayoutOrder = 2 })
	end
	tween(card, 0.3, { Position = UDim2.fromOffset(0, 0) }, Enum.EasingStyle.Quint)
	task.delay(cfg.Duration or 4, function()
		if not card.Parent then return end
		local tw = tween(card, 0.25, { Position = UDim2.fromOffset(270, 0) }, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
		tw.Completed:Connect(function() wrap:Destroy() end)
	end)
end

function Nauru:AddTheme(name, t)
	name = tostring(name)
	t = finishTheme(table.clone(t))
	if not THEMES[name] then table.insert(self.ThemeOrder, name) end
	THEMES[name] = t
end

function Nauru:SetTheme(name)
	local t = THEMES[name]
	if not t then return false end
	self.Theme = name
	for k, v in pairs(t) do C[k] = v end
	for i = #bound, 1, -1 do
		local b = bound[i]
		if not b[1].Parent then
			table.remove(bound, i)
		else
			for prop, key in pairs(b[2]) do b[1][prop] = C[key] end
		end
	end
	for _, h in ipairs(hooks) do pcall(h) end
	return true
end

local function keyName(v)
	if v == nil then return "NONE" end
	if v.EnumType == Enum.KeyCode then return string.sub(string.upper(v.Name), 1, 6) end
	local map = { MouseButton1 = "M1", MouseButton2 = "M2", MouseButton3 = "M3" }
	return map[v.Name] or v.Name
end

local function parseEnum(s)
	if type(s) ~= "string" then return nil end
	local t, n = string.match(s, "^Enum%.(%w+)%.(%w+)$")
	if t and n and Enum[t] and Enum[t][n] then return Enum[t][n] end
	return nil
end

function Nauru:CreateWindow(cfg)
	cfg = cfg or {}
	ensureGui()
	if cfg.Theme and THEMES[cfg.Theme] then self:SetTheme(cfg.Theme) end

	local W = cfg.Size and cfg.Size.X or 540
	local H = cfg.Size and cfg.Size.Y or 380
	local win = setmetatable({ Tabs = {}, Key = cfg.Key or Enum.KeyCode.RightShift, Visible = true, Folder = cfg.Folder or "NauruUI", _conns = {} }, Window)
	table.insert(self.Windows, win)

	local holder = frame(gui, { Name = "Window", AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5), Size = UDim2.fromOffset(W, H) })
	local scale = create("UIScale", {}, holder)
	win.Holder = holder

	local shadows = {}
	for i = 1, 4 do
		local s = frame(holder, { Position = UDim2.fromOffset(-i * 3, -i * 3 + 5), Size = UDim2.new(1, i * 6, 1, i * 6), BackgroundColor3 = Color3.new(0, 0, 0), BackgroundTransparency = 0.93, ZIndex = 0 })
		corner(s, 10 + i * 3)
		table.insert(shadows, s)
	end

	local panel = create("CanvasGroup", { AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5), Size = UDim2.fromScale(1, 1), BorderSizePixel = 0 }, holder)
	bind(panel, { BackgroundColor3 = "bg" })
	corner(panel, 10)
	stroke(panel, "accent", 1, 0.8)
	local pop = create("UIScale", { Scale = 1 }, panel)

	-- responsive scale: the window never leaves the screen
	local function fit()
		local vp = gui.AbsoluteSize
		scale.Scale = math.clamp(math.min((vp.X - 24) / W, (vp.Y - 24) / H, 1), 0.45, 1)
	end
	fit()
	gui:GetPropertyChangedSignal("AbsoluteSize"):Connect(fit)

	-- header
	local header = frame(panel, { Size = UDim2.new(1, 0, 0, 40) })
	local title = label(header, "", 15, "text", true, { Position = UDim2.fromOffset(14, 0), Size = UDim2.new(1, -60, 1, 0), RichText = true })
	local function brand()
		title.Text = string.format('<font color="%s">%s</font><font color="%s"> %s</font>', rgbStr(C.accent), cfg.Title or "NAURU", rgbStr(C.red), cfg.Subtitle or "UI")
	end
	brand()
	table.insert(hooks, brand)
	local line = frame(panel, { Position = UDim2.fromOffset(0, 40), Size = UDim2.new(1, 0, 0, 1) })
	bind(line, { BackgroundColor3 = "line", BackgroundTransparency = nil })
	line.BackgroundTransparency = 0

	local closeBtn = button(header, { AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -10, 0.5, 0), Size = UDim2.fromOffset(24, 24) })
	for _, rot in ipairs({ 45, -45 }) do
		local bar = frame(closeBtn, { AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5), Size = UDim2.fromOffset(12, 1.5), Rotation = rot, BackgroundTransparency = 0 })
		bind(bar, { BackgroundColor3 = "sub" })
		corner(bar, 1)
	end
	closeBtn.MouseEnter:Connect(function() tween(closeBtn, 0.1, { BackgroundTransparency = 0.85, BackgroundColor3 = C.red }) end)
	closeBtn.MouseLeave:Connect(function() tween(closeBtn, 0.1, { BackgroundTransparency = 1 }) end)
	corner(closeBtn, 6)
	closeBtn.Activated:Connect(function() win:Toggle(false) end)

	-- sidebar + content
	local side = frame(panel, { Position = UDim2.fromOffset(0, 41), Size = UDim2.new(0, 124, 1, -41) })
	create("UIPadding", { PaddingTop = UDim.new(0, 8), PaddingLeft = UDim.new(0, 8), PaddingRight = UDim.new(0, 8) }, side)
	create("UIListLayout", { Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder }, side)
	local sideLine = frame(panel, { Position = UDim2.fromOffset(124, 41), Size = UDim2.new(0, 1, 1, -41), BackgroundTransparency = 0 })
	bind(sideLine, { BackgroundColor3 = "line" })
	local content = frame(panel, { Position = UDim2.fromOffset(125, 41), Size = UDim2.new(1, -125, 1, -41) })
	win.Side, win.Content = side, content

	-- drag by header
	do
		local dragging, startPos, startMouse
		header.InputBegan:Connect(function(input)
			if isPress(input) then
				dragging, startPos, startMouse = true, holder.Position, input.Position
			end
		end)
		UserInputService.InputChanged:Connect(function(input)
			if dragging and isMove(input) then
				local d = input.Position - startMouse
				local vp = gui.AbsoluteSize
				local nx = math.clamp(startPos.X.Offset + d.X, -vp.X / 2 + 60, vp.X / 2 - 60)
				local ny = math.clamp(startPos.Y.Offset + d.Y, -vp.Y / 2 + 20, vp.Y / 2 - 40)
				holder.Position = UDim2.new(startPos.X.Scale, nx, startPos.Y.Scale, ny)
			end
		end)
		UserInputService.InputEnded:Connect(function(input) if isPress(input) then dragging = false end end)
	end

	-- small open button (always created; shows only while the window is hidden)
	local openBtn
	if cfg.OpenButton ~= false then
		openBtn = button(gui, { Name = "NauruOpen", AnchorPoint = Vector2.new(0.5, 0), Position = UDim2.new(0.5, 0, 0, 8), Size = UDim2.fromOffset(34, 34), BackgroundTransparency = 0, Visible = false, Text = "N", Font = FONT_BOLD, TextSize = 16 })
		bind(openBtn, { BackgroundColor3 = "panel", TextColor3 = "accent" })
		corner(openBtn, 17)
		stroke(openBtn, "accent", 1, 0.5)
		openBtn.Activated:Connect(function() win:Toggle(true) end)
	end

	function win:Toggle(v)
		if v == nil then v = not self.Visible end
		if v == self.Visible then return end
		self.Visible = v
		if openBtn then openBtn.Visible = not v end
		if v then
			holder.Visible = true
			pop.Scale = 0.92
			panel.GroupTransparency = 1
			tween(pop, 0.22, { Scale = 1 }, Enum.EasingStyle.Back)
			tween(panel, 0.18, { GroupTransparency = 0 })
			for _, s in ipairs(shadows) do tween(s, 0.2, { BackgroundTransparency = 0.93 }) end
		else
			tween(pop, 0.15, { Scale = 0.94 })
			tween(panel, 0.15, { GroupTransparency = 1 })
			for _, s in ipairs(shadows) do tween(s, 0.15, { BackgroundTransparency = 1 }) end
			task.delay(0.16, function() if not self.Visible then holder.Visible = false end end)
		end
	end

	function win:SetKey(k) self.Key = k end

	table.insert(win._conns, UserInputService.InputBegan:Connect(function(input, gpe)
		if capturing or UserInputService:GetFocusedTextBox() then return end
		if input.KeyCode == win.Key and input.KeyCode ~= Enum.KeyCode.Unknown then win:Toggle() end
	end))

	function win:Destroy()
		for _, c in ipairs(self._conns) do c:Disconnect() end
		holder:Destroy()
		if openBtn then openBtn:Destroy() end
		for i, w in ipairs(Nauru.Windows) do if w == self then table.remove(Nauru.Windows, i) break end end
	end

	holder.Visible = true
	pop.Scale = 0.92
	panel.GroupTransparency = 1
	tween(pop, 0.28, { Scale = 1 }, Enum.EasingStyle.Back)
	tween(panel, 0.2, { GroupTransparency = 0 })
	return win
end

function Window:AddTab(name)
	local tab = setmetatable({ Window = self, Name = name }, Tab)
	local idx = #self.Tabs + 1

	local page = create("ScrollingFrame", {
		Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, BorderSizePixel = 0, Visible = false,
		ScrollBarThickness = 2, CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.None, ScrollingDirection = Enum.ScrollingDirection.Y,
	}, self.Content)
	bind(page, { ScrollBarImageColor3 = "accent" })
	create("UIPadding", { PaddingTop = UDim.new(0, 12), PaddingBottom = UDim.new(0, 12), PaddingLeft = UDim.new(0, 14), PaddingRight = UDim.new(0, 16) }, page)
	local pageLayout = create("UIListLayout", { Padding = UDim.new(0, 8), SortOrder = Enum.SortOrder.LayoutOrder }, page)
	local function fitCanvas() -- exact content height + the 12px top / bottom padding, so you can't scroll past the last element
		page.CanvasSize = UDim2.fromOffset(0, pageLayout.AbsoluteContentSize.Y + 24)
	end
	pageLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(fitCanvas)
	fitCanvas()
	tab.page = page

	local btn = button(self.Side, { Size = UDim2.new(1, 0, 0, 28), LayoutOrder = idx, BackgroundTransparency = 1 })
	bind(btn, { BackgroundColor3 = "accent" })
	corner(btn, 6)
	local bar = frame(btn, { AnchorPoint = Vector2.new(0, 0.5), Position = UDim2.new(0, 0, 0.5, 0), Size = UDim2.fromOffset(3, 0), BackgroundTransparency = 0 })
	bind(bar, { BackgroundColor3 = "accent" })
	corner(bar, 2)
	local text = label(btn, name, 12, "dim", true, { Position = UDim2.fromOffset(14, 0), Size = UDim2.new(1, -18, 1, 0) })
	tab.button, tab.bar, tab.text = btn, bar, text

	btn.MouseEnter:Connect(function() if self.Current ~= tab then tween(btn, 0.1, { BackgroundTransparency = 0.94 }) end end)

	self.Current = tab
	for _, t in ipairs(self.Tabs) do

		tween(t.text, d, { TextColor3 = on and C.text or C.dim })
		tween(t.button, d, { BackgroundTransparency = on and 0.88 or 1 })
	end
end

--------------------------------------------------------------------------
-- CONFIGS (needs writefile / readfile)
--------------------------------------------------------------------------
local function fsOk() return writefile and readfile and isfile and isfolder and makefolder and true or false end

local function cfgPath(win, name) return win.Folder .. "/" .. name .. ".json" end

function Window:SaveConfig(name)
	if not fsOk() then return false, "no file access" end
	name = (name and name ~= "") and name or "default"
	local data = {}
	for flag, h in pairs(Nauru.Flags) do
		if h._save then
			local ok, v = pcall(h._save)
			if ok then data[flag] = v end
		end
	end
	if not isfolder(self.Folder) then makefolder(self.Folder) end
	writefile(cfgPath(self, name), HttpService:JSONEncode(data))
	return true
end

function Window:LoadConfig(name)
	if not fsOk() then return false, "no file access" end
	name = (name and name ~= "") and name or "default"
	local p = cfgPath(self, name)
	if not isfile(p) then return false, "config not found" end
	local ok, data = pcall(function() return HttpService:JSONDecode(readfile(p)) end)
	if not ok or type(data) ~= "table" then return false, "bad config file" end
	for flag, v in pairs(data) do
		local h = Nauru.Flags[flag]
		if h and h._load then pcall(h._load, v) end
	end
	return true
end

function Window:DeleteConfig(name)
	if delfile and isfile and isfile(cfgPath(self, name)) then delfile(cfgPath(self, name)) return true end
	return false
end

function Window:ListConfigs()
	local out = {}
	if listfiles and isfolder and isfolder(self.Folder) then
		for _, f in ipairs(listfiles(self.Folder)) do
			local n = string.match(f, "([^/\\]+)%.json$")
			if n then table.insert(out, n) end
		end
	end
	table.sort(out)
	return out
end

	t:AddSection("Interface")
	t:AddDropdown({ Name = "Theme", Options = Nauru.ThemeOrder, Default = Nauru.Theme, Callback = function(v) Nauru:SetTheme(v) end })
	t:AddKeybind({ Name = "Menu key", Default = self.Key, Changed = function(k) if k then self:SetKey(k) end end })
	t:AddSection("Configs")
	local pickedName = "default"
	local list
	t:AddInput({ Name = "Config name", Placeholder = "config name", Default = "default", Callback = function(text) pickedName = text ~= "" and text or "default" end })
	list = t:AddDropdown({ Name = "Saved configs", Options = #self:ListConfigs() > 0 and self:ListConfigs() or { "-" }, Default = "-", Callback = function(v) if v ~= "-" then pickedName = v end end })
	t:AddButton({ Name = "Save config", Filled = true, Callback = function()
		local ok, err = self:SaveConfig(pickedName)
		Nauru:Notify({ Title = ok and "Config saved" or "Save failed", Text = ok and pickedName or tostring(err), Duration = 3 })
		local l = self:ListConfigs()
		list:Refresh(#l > 0 and l or { "-" }, pickedName)
	end })
	t:AddButton({ Name = "Load config", Callback = function()
		local ok, err = self:LoadConfig(pickedName)
		Nauru:Notify({ Title = ok and "Config loaded" or "Load failed", Text = ok and pickedName or tostring(err), Duration = 3 })
	end })
	t:AddButton({ Name = "Delete config", Callback = function()
		local ok = self:DeleteConfig(pickedName)
		Nauru:Notify({ Title = ok and "Config deleted" or "Delete failed", Text = pickedName, Duration = 3 })
		local l = self:ListConfigs()
		list:Refresh(#l > 0 and l or { "-" }, l[1] or "-")
	end })
	return t
end

function Nauru:Destroy()
	for i = #self.Windows, 1, -1 do self.Windows[i]:Destroy() end
	if gui then gui:Destroy() gui = nil end
end

return Nauru
